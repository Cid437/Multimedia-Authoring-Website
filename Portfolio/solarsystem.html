# Python script used to build the Solar System
import bpy
import math
import random

# =========================================================
# CLEAN SCENE
# =========================================================
bpy.ops.object.select_all(action='SELECT')
bpy.ops.object.delete(use_global=False)

# =========================================================
# RENDER SETTINGS — 640x360 optimized
# =========================================================
scene = bpy.context.scene
scene.render.resolution_x = 640
scene.render.resolution_y = 360
scene.render.resolution_percentage = 100
scene.render.fps = 24
scene.render.engine = 'CYCLES'
scene.cycles.samples = 64
scene.cycles.use_denoising = True
scene.render.film_transparent = False

# =========================================================
# SETTINGS
# =========================================================
SCALE_DISTANCE      = 2.2
PLANET_SIZE_MULT    = 1.2
TEXTURE_FOLDER      = r"C:\BSIT-FILES\2Y3S\Multimedia&Authoring\solarsystem\planettextures\\"

FRAME_END           = 1440
TOUR_START          = 120
FRAMES_PER_PLANET   = 90
PULLBACK_START      = TOUR_START + 8 * FRAMES_PER_PLANET
PULLBACK_END        = PULLBACK_START + 120
SYSTEM_START        = PULLBACK_END

# =========================================================
# PLANET DATA
# =========================================================
planets = [
    ("Mercury", 0.45,  4,  4.15, 0.8,  "mercury.jpg",   0,    False),
    ("Venus",   0.70,  6,  1.62, 0.6,  "venus.jpg",   177,    True),
    ("Earth",   0.75,  8,  1.00, 1.5,  "earth.jpg",   23.5,   False),
    ("Mars",    0.60, 10,  0.53, 1.4,  "mars.jpg",    25,     False),
    ("Jupiter", 2.30, 14,  0.084,3.5,  "jupiter.jpg",   3,    False),
    ("Saturn",  2.00, 18,  0.034,3.2,  "saturn.jpg",  27,     False),
    ("Uranus",  1.30, 22,  0.012,2.0,  "uranus.jpg",  98,     True),
    ("Neptune", 1.25, 26,  0.006,2.2,  "neptune.jpg", 28,     False),
]

# =========================================================
# HELPER: ease-in-out interpolation for keyframes
# =========================================================
def set_easing(obj, data_path, index=-1):
    if index == -1:
        fcurves = obj.animation_data.action.fcurves
    else:
        fcurves = [obj.animation_data.action.fcurves.find(data_path, index=index)]
    for fc in fcurves:
        if fc is None:
            continue
        for kp in fc.keyframe_points:
            kp.interpolation = 'BEZIER'
            kp.easing = 'EASE_IN_OUT'

# =========================================================
# SUN
# =========================================================
bpy.ops.mesh.primitive_uv_sphere_add(
    radius=3, segments=64, ring_count=32, location=(0, 0, 0)
)
sun = bpy.context.object
sun.name = "Sun"
bpy.ops.object.shade_smooth()

sun_mat = bpy.data.materials.new(name="SunMaterial")
sun_mat.use_nodes = True
nodes = sun_mat.node_tree.nodes
links = sun_mat.node_tree.links
nodes.clear()

tex_node  = nodes.new(type="ShaderNodeTexImage")
emission  = nodes.new(type="ShaderNodeEmission")
out_node  = nodes.new(type="ShaderNodeOutputMaterial")

try:
    tex_node.image = bpy.data.images.load(TEXTURE_FOLDER + "sunmap.jpg", check_existing=True)
    links.new(tex_node.outputs["Color"], emission.inputs["Color"])
except Exception:
    emission.inputs["Color"].default_value = (1.0, 0.6, 0.1, 1.0)

links.new(emission.outputs["Emission"], out_node.inputs["Surface"])
emission.inputs["Strength"].default_value = 12
sun.data.materials.append(sun_mat)

bpy.ops.object.light_add(type='POINT', location=(0, 0, 0))
sun_light = bpy.context.object
sun_light.name = "SunLight"
sun_light.data.energy       = 8000
sun_light.data.shadow_soft_size = 3.0
sun_light.data.use_shadow   = True

bpy.ops.object.light_add(type='SUN', location=(0, 0, 50))
fill_light = bpy.context.object
fill_light.name = "AmbientFill"
fill_light.data.energy  = 0.04
fill_light.data.angle   = math.radians(180)
fill_light.data.use_shadow = False

drv = sun.driver_add("rotation_euler", 2).driver
drv.expression = f"frame * 0.01"

# =========================================================
# CREATE PLANETS + COLLECT POSITIONS FOR CAMERA TOUR
# =========================================================
planet_objects = {}
orbit_empties  = {}

for name, size, distance, orbit_speed, rot_speed, texture, tilt, retrograde in planets:

    bpy.ops.object.empty_add(type='PLAIN_AXES', location=(0, 0, 0))
    orbit_empty = bpy.context.object
    orbit_empty.name = name + "_Orbit"
    orbit_empties[name] = orbit_empty

    bpy.ops.object.select_all(action='DESELECT')
    bpy.ops.mesh.primitive_uv_sphere_add(
        radius=size * PLANET_SIZE_MULT,
        segments=64,
        ring_count=32,
        location=(distance * SCALE_DISTANCE, 0, 0)
    )
    planet = bpy.context.object
    planet.name = name
    bpy.ops.object.shade_smooth()
    planet_objects[name] = planet

    mat = bpy.data.materials.new(name=name + "_Material")
    mat.use_nodes = True
    pnodes = mat.node_tree.nodes
    plinks = mat.node_tree.links
    pnodes.clear()

    p_tex  = pnodes.new(type="ShaderNodeTexImage")
    p_emit = pnodes.new(type="ShaderNodeEmission")
    p_out  = pnodes.new(type="ShaderNodeOutputMaterial")

    try:
        p_tex.image = bpy.data.images.load(TEXTURE_FOLDER + texture, check_existing=True)
        plinks.new(p_tex.outputs["Color"], p_emit.inputs["Color"])
    except Exception:
        p_emit.inputs["Color"].default_value = (0.5, 0.5, 0.5, 1.0)

    plinks.new(p_emit.outputs["Emission"], p_out.inputs["Surface"])
    p_emit.inputs["Strength"].default_value = 1.5
    planet.data.materials.append(mat)

    planet.parent = orbit_empty
    planet.matrix_parent_inverse = orbit_empty.matrix_world.inverted()

    planet.rotation_euler[0] = math.radians(tilt)

    rot_dir = -1 if retrograde else 1
    drv_spin = planet.driver_add("rotation_euler", 2).driver
    drv_spin.expression = f"frame * {rot_dir * rot_speed} * 0.05"

    drv_orb = orbit_empty.driver_add("rotation_euler", 2).driver
    drv_orb.expression = (
        f"{orbit_speed} * 80 * (1 - cos(frame / {FRAME_END} * 1.5708))"
    )

    if name == "Saturn":
        bpy.ops.object.select_all(action='DESELECT')
        bpy.ops.mesh.primitive_torus_add(
            align='WORLD',
            location=(distance * SCALE_DISTANCE, 0, 0),
            major_radius=3.8,
            minor_radius=0.55,
            major_segments=80,
            minor_segments=12
        )
        ring = bpy.context.object
        ring.name = "SaturnRing"
        ring.scale[2]         = 0.08
        ring.rotation_euler[0] = math.radians(27)
        ring.parent           = orbit_empty
        ring.matrix_parent_inverse = orbit_empty.matrix_world.inverted()

        ring_mat   = bpy.data.materials.new(name="SaturnRing_Material")
        ring_mat.use_nodes = True
        rn = ring_mat.node_tree.nodes
        rl = ring_mat.node_tree.links
        rn.clear()

        r_emit = rn.new(type="ShaderNodeEmission")
        r_out  = rn.new(type="ShaderNodeOutputMaterial")

        try:
            r_tex       = rn.new(type="ShaderNodeTexImage")
            r_tex.image = bpy.data.images.load(TEXTURE_FOLDER + "saturn_ring.png", check_existing=True)
            rl.new(r_tex.outputs["Color"], r_emit.inputs["Color"])
        except Exception:
            r_emit.inputs["Color"].default_value = (0.85, 0.72, 0.50, 1.0)

        r_emit.inputs["Strength"].default_value = 1.2
        rl.new(r_emit.outputs["Emission"], r_out.inputs["Surface"])
        ring.data.materials.append(ring_mat)

# =========================================================
# EARTH MOON
# =========================================================
earth_obj   = planet_objects["Earth"]
earth_orbit = orbit_empties["Earth"]
earth_dist  = 8 * SCALE_DISTANCE

bpy.ops.object.empty_add(type='PLAIN_AXES', location=(earth_dist, 0, 0))
moon_orbit_empty = bpy.context.object
moon_orbit_empty.name = "Moon_Orbit"
moon_orbit_empty.parent = earth_orbit
moon_orbit_empty.matrix_parent_inverse = earth_orbit.matrix_world.inverted()

bpy.ops.object.select_all(action='DESELECT')
bpy.ops.mesh.primitive_uv_sphere_add(
    radius=0.22,
    segments=32,
    ring_count=16,
    location=(earth_dist + 1.8, 0, 0)
)
moon = bpy.context.object
moon.name = "Moon"
bpy.ops.object.shade_smooth()

moon_mat   = bpy.data.materials.new(name="Moon_Material")
moon_mat.use_nodes = True
mn = moon_mat.node_tree.nodes
ml = moon_mat.node_tree.links
mn.clear()

m_tex  = mn.new(type="ShaderNodeTexImage")
m_emit = mn.new(type="ShaderNodeEmission")
m_out  = mn.new(type="ShaderNodeOutputMaterial")

try:
    m_tex.image = bpy.data.images.load(TEXTURE_FOLDER + "moon.jpg", check_existing=True)
    ml.new(m_tex.outputs["Color"], m_emit.inputs["Color"])
except Exception:
    m_emit.inputs["Color"].default_value = (0.75, 0.75, 0.75, 1.0)

ml.new(m_emit.outputs["Emission"], m_out.inputs["Surface"])
m_emit.inputs["Strength"].default_value = 0.8
moon.data.materials.append(moon_mat)

moon.parent = moon_orbit_empty
moon.matrix_parent_inverse = moon_orbit_empty.matrix_world.inverted()

drv_moon = moon_orbit_empty.driver_add("rotation_euler", 2).driver
drv_moon.expression = "frame * 0.25"

# =========================================================
# ORBIT VISUALIZATION RINGS
# =========================================================
for name, size, distance, orbit_speed, rot_speed, texture, tilt, retrograde in planets:
    bpy.ops.curve.primitive_bezier_circle_add(
        radius=distance * SCALE_DISTANCE,
        location=(0, 0, 0)
    )
    orbit_vis = bpy.context.object
    orbit_vis.name = name + "_OrbitPath"
    orbit_vis.data.bevel_depth      = 0.015
    orbit_vis.data.bevel_resolution = 2
    orbit_vis.data.resolution_u    = 64

    orbit_mat = bpy.data.materials.new(name=name + "_OrbitMat")
    orbit_mat.use_nodes = True
    on = orbit_mat.node_tree.nodes
    ol = orbit_mat.node_tree.links
    on.clear()
    o_emit = on.new(type="ShaderNodeEmission")
    o_out  = on.new(type="ShaderNodeOutputMaterial")
    o_emit.inputs["Color"].default_value   = (0.3, 0.5, 0.8, 1.0)
    o_emit.inputs["Strength"].default_value = 0.3
    ol.new(o_emit.outputs["Emission"], o_out.inputs["Surface"])
    orbit_vis.data.materials.append(orbit_mat)

# =========================================================
# SHOOTING STARS
# =========================================================
random.seed(42)
shooting_stars = []

for i in range(20):
    angle   = random.uniform(0, 2 * math.pi)
    radius  = random.uniform(35, 70)
    z_start = random.uniform(-10, 10)
    x_start = math.cos(angle) * radius
    y_start = math.sin(angle) * radius

    dx = random.uniform(-1, 1) * 30
    dy = random.uniform(-1, 1) * 30
    dz = random.uniform(-3, 3)

    bpy.ops.object.select_all(action='DESELECT')
    bpy.ops.mesh.primitive_uv_sphere_add(
        radius=0.08, segments=8, ring_count=4,
        location=(x_start, y_start, z_start)
    )
    star = bpy.context.object
    star.name = f"ShootingStar_{i:02d}"
    star.scale = (1, 1, 5)

    travel_vec = (dx, dy, dz)
    length     = math.sqrt(sum(v**2 for v in travel_vec))
    if length > 0:
        nx, ny, nz = [v / length for v in travel_vec]
        dot   = nz
        angle_to_z = math.acos(max(-1, min(1, dot)))
        rx = -ny
        ry =  nx
        rlen = math.sqrt(rx**2 + ry**2)
        if rlen > 0.001:
            rx /= rlen
            ry /= rlen
            star.rotation_euler = (rx * angle_to_z, ry * angle_to_z, 0)

    ss_mat = bpy.data.materials.new(name=f"ShootingStar_{i:02d}_Mat")
    ss_mat.use_nodes = True
    sn = ss_mat.node_tree.nodes
    sl = ss_mat.node_tree.links
    sn.clear()
    s_emit = sn.new(type="ShaderNodeEmission")
    s_out  = sn.new(type="ShaderNodeOutputMaterial")
    s_emit.inputs["Color"].default_value    = (1.0, 0.95, 0.85, 1.0)
    s_emit.inputs["Strength"].default_value = 6.0
    sl.new(s_emit.outputs["Emission"], s_out.inputs["Surface"])
    star.data.materials.append(ss_mat)

    fire_frame  = random.randint(1, FRAME_END - 60)
    exist_frames = random.randint(18, 36)

    star.hide_render = True
    star.hide_viewport = True
    star.keyframe_insert(data_path="hide_render",   frame=1)
    star.keyframe_insert(data_path="hide_viewport", frame=1)

    star.hide_render = False
    star.hide_viewport = False
    star.keyframe_insert(data_path="hide_render",   frame=fire_frame)
    star.keyframe_insert(data_path="hide_viewport", frame=fire_frame)

    star.location = (x_start, y_start, z_start)
    star.keyframe_insert(data_path="location", frame=fire_frame)

    star.location = (x_start + dx, y_start + dy, z_start + dz)
    star.keyframe_insert(data_path="location", frame=fire_frame + exist_frames)

    star.hide_render = True
    star.hide_viewport = True
    star.keyframe_insert(data_path="hide_render",   frame=fire_frame + exist_frames + 1)
    star.keyframe_insert(data_path="hide_viewport", frame=fire_frame + exist_frames + 1)

    shooting_stars.append(star)

world = bpy.context.scene.world

try:
    w_env.image = bpy.data.images.load(TEXTURE_FOLDER + "2k_stars.jpg", check_existing=True)
    wl.new(w_env.outputs["Color"], w_bg.inputs["Color"])
except Exception:

wl.new(w_bg.outputs["Background"], w_out.inputs["Surface"])
w_bg.inputs["Strength"].default_value = 1.3

# =========================================================
# CAMERA RIG
# =========================================================
bpy.ops.object.camera_add(location=(0, -10, 6))
camera = bpy.context.object
camera.name = "CinematicCamera"
camera.data.clip_end    = 2000
camera.data.lens        = 35
bpy.context.scene.camera = camera

def cam_key(frame, loc, rot_deg):
    bpy.context.scene.frame_set(frame)
    camera.location       = loc
    camera.rotation_euler = (math.radians(rot_deg[0]),
                             math.radians(rot_deg[1]),
                             math.radians(rot_deg[2]))
    camera.keyframe_insert(data_path="location")
    camera.keyframe_insert(data_path="rotation_euler")

cam_key(1,   (0, -8,   3),   (85, 0, 0))
cam_key(60,  (0, -12,  5),   (80, 0, 0))
cam_key(120, (0, -18,  8),   (75, 0, 0))

tour_shots = [
    ("Mercury", 10,  -5,   2,   80),
    ("Venus",   14,  -6,   3,   78),
    ("Earth",   18,  -8,   4,   75),
    ("Mars",    23,  -9,   4,   75),
    ("Jupiter", 32, -14,   8,   72),
    ("Saturn",  40, -16,  10,   70),
    ("Uranus",  49, -18,  10,   70),
    ("Neptune", 58, -20,  12,   68),
]

for idx, (pname, cx, cy, cz, tx) in enumerate(tour_shots):
    f_in  = TOUR_START + idx * FRAMES_PER_PLANET
    f_mid = f_in + FRAMES_PER_PLANET // 2
    f_out = f_in + FRAMES_PER_PLANET

    cam_key(f_in,  (cx - 3, cy,     cz),     (tx, 0, -5))
    cam_key(f_mid, (cx,     cy + 2, cz + 1), (tx - 2, 0, 0))
    cam_key(f_out, (cx + 3, cy - 2, cz + 1), (tx - 3, 0, 3))

cam_key(PULLBACK_START, (60, -22, 14), (68, 0, 5))
cam_key(PULLBACK_START + 60, (0, -80, 45), (62, 0, 0))
cam_key(PULLBACK_END,   (0, -100, 55), (58, 0, 0))

orbit_radius = 105
for t_idx, angle_deg in enumerate(range(0, 181, 30)):
    fframe = SYSTEM_START + t_idx * (480 // 6)
    a      = math.radians(angle_deg)
    cam_x  = math.sin(a) * orbit_radius
    cam_y  = -math.cos(a) * orbit_radius
    cam_z  = 55 + t_idx * 2
    cam_key(fframe, (cam_x, cam_y, cam_z), (58, 0, angle_deg))

cam_key(FRAME_END, (0, -110, 90), (52, 0, 0))

if camera.animation_data and camera.animation_data.action:
    for fc in camera.animation_data.action.fcurves:
        for kp in fc.keyframe_points:
            kp.interpolation = 'BEZIER'
            kp.easing = 'EASE_IN_OUT'

# =========================================================
# DEPTH OF FIELD

camera.data.dof.use_dof     = True
camera.data.dof.focus_distance = 20.0
camera.data.dof.aperture_fstop = 4.0

# =========================================================
